local tabs = {}
local tabButtons = {}

local function createTabButton(name, displayName, order)
    local btn = Instance.new("TextButton")
    btn.LayoutOrder = order
    btn.Size = UDim2.new(1, 0, 0, 35)
    btn.BackgroundColor3 = CARD_BG
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Text = "  " .. displayName
    btn.Font = Enum.Font.SourceSansBold
    btn.TextSize = 13
    btn.TextXAlignment = Enum.TextXAlignment.Left

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)  
      
    local indicator = Instance.new("Frame")  
    indicator.Size = UDim2.new(0, 3, 0, 20)  
    indicator.Position = UDim2.new(0, 4, 0.5, -10)  
    indicator.BackgroundColor3 = GREY_ACCENT  
    indicator.Visible = false  
    indicator.Parent = btn  
    Instance.new("UICorner", indicator).CornerRadius = UDim.new(0, 2)  
      
    local stroke = Instance.new("UIStroke")  
    stroke.Color = GREY_BORDER  
    stroke.Thickness = 1  
    stroke.Parent = btn  
      
    btn.Parent = sidebarFrame  
      
    local page = Instance.new("Frame")  
    page.Size = UDim2.new(1, 0, 1, 0)  
    page.BackgroundTransparency = 1  
    page.Visible = false  
    page.Parent = contentFrame  
      
    tabs[name] = page  
    tabButtons[name] = {Button = btn, Stroke = stroke, Indicator = indicator}  
      
    btn.MouseButton1Click:Connect(function()  
        for tabName, tabData in pairs(tabButtons) do  
            local isSelected = (tabName == name)  
            tabs[tabName].Visible = isSelected  
            tabData.Indicator.Visible = isSelected  
            if isSelected then  
                tabData.Button.BackgroundColor3 = BUTTON_BG  
                tabData.Stroke.Color = GREY_ACCENT  
            else  
                tabData.Button.BackgroundColor3 = CARD_BG  
                tabData.Stroke.Color = GREY_BORDER  
            end  
        end  
    end)  
    return page
end

-- Tabs Setup in exact required order:
local mainPage         = createTabButton("Main",         "1. Main",         1)
local coordinationPage = createTabButton("Coordination", "2. Coordination", 2)
local teleportPage     = createTabButton("Teleport",     "3. Teleport",     3)

local function applyBtnStyle(btn)
    local s = Instance.new("UIStroke")
    s.Color = GREY_ACCENT
    s.Thickness = 1
    s.Transparency = 0.5
    s.Parent = btn
    return s
end

local function parseCoordsString(str)
    if not str or str == "" then return nil end
    local x, y, z = string.match(str, "([%d.-]+)%s*,%s*([%d.-]+)%s*,%s*([%d.-]+)")
    if x and y and z then
        return Vector3.new(tonumber(x), tonumber(y), tonumber(z))
    end
    return nil
end

    if (isAutoFlyOn or isNoclipOn) and player.Character then

autoFlyToggleBtn.MouseButton1Click:Connect(function()
    isAutoFlyOn = not isAutoFlyOn
    autoFlyToggleBtn.Text = isAutoFlyOn and "Auto Fly: ON" or "Auto Fly: OFF"
    if isAutoFlyOn then
        task.spawn(function()
            while isAutoFlyOn do
                local didFly = false
                for i = 1, #waypointsData do
                    if not isAutoFlyOn then break end
                    local targetPos = parseCoordsString(waypointsData[i].Input.Text)
                    if targetPos then
                        didFly = true
                        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                            local hrp = player.Character.HumanoidRootPart
                            local spd = tonumber(speedInput.Text) or 500
                            local dist = (hrp.Position - targetPos).Magnitude
                            local tweenInfo = TweenInfo.new(dist / spd, Enum.EasingStyle.Linear)
                            currentTween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.new(targetPos)})
                            local bv = Instance.new("BodyVelocity")
                            bv.Velocity = Vector3.new(0,0,0)
                            bv.MaxForce = Vector3.new(9e9,9e9,9e9)
                            bv.Parent = hrp
                            currentTween:Play()
                            currentTween.Completed:Wait()
                            if bv then bv:Destroy() end
                        end
                    end
                end
                if isAutoFlyOn and didFly then task.wait(1.5) elseif not didFly then isAutoFlyOn = false autoFlyToggleBtn.Text = "Auto Fly: OFF" break end
            end
        end)
    else
        if currentTween then currentTween:Cancel() end
    end
end)
