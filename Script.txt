local logsContainer = nil

local function addLog(text)
    local timeStr = os.date("%H:%M:%S")
    local logText = "[" .. timeStr .. "] " .. text
    
    -- Print to Logs Tab if exists
    if logsContainer then
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(1, -10, 0, 20)
        lbl.BackgroundTransparency = 1
        lbl.Text = logText
        lbl.TextColor3 = GREY_ACCENT
        lbl.Font = Enum.Font.SourceSans
        lbl.TextSize = 12
        lbl.TextXAlignment = Enum.TextXAlignment.Left
        lbl.Parent = logsContainer
        
        -- Auto scroll to bottom
        if logsContainer.Parent:IsA("ScrollingFrame") then
            logsContainer.Parent.CanvasPosition = Vector2.new(0, 99999)
        end
    end
    
    -- Show Toast Notification
    if enableToasts then
        local card = Instance.new("Frame")
        card.Size = UDim2.new(1, 0, 0, 30)
        card.BackgroundColor3 = CARD_BG
        card.Parent = notificationContainer
        Instance.new("UICorner", card).CornerRadius = UDim.new(0, 6)
        local stroke = Instance.new("UIStroke")
        stroke.Color = GREY_ACCENT
        stroke.Thickness = 1
        stroke.Transparency = 0.3
        stroke.Parent = card
        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(1, -10, 1, 0)
        lbl.Position = UDim2.new(0, 5, 0, 0)
        lbl.BackgroundTransparency = 1
        lbl.Text = text
        lbl.TextColor3 = TEXT_MAIN
        lbl.Font = Enum.Font.SourceSansBold
        lbl.TextSize = 12
        lbl.Parent = card
        task.delay(2.5, function() if card then card:Destroy() end end)
    end
end

local function createButtonAnim(btn)
    btn.MouseEnter:Connect(function() TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = HOVER_BG}):Play() end)
    btn.MouseLeave:Connect(function() TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = BUTTON_BG}):Play() end)
end

local function applyBtnStyle(btn)
    local s = Instance.new("UIStroke")
    s.Color = GREY_ACCENT; s.Thickness = 1; s.Transparency = 0.6; s.Parent = btn
    createButtonAnim(btn)
end

local function createCard(parent, height)

local function createToggle(parent, title, desc, defaultState, callback)
    local card = createCard(parent, 50)
    
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0, 200, 0, 20)
    lbl.Position = UDim2.new(0, 12, 0, 6)
    lbl.BackgroundTransparency = 1; lbl.Text = title; lbl.TextColor3 = TEXT_MAIN
    lbl.Font = Enum.Font.SourceSansBold; lbl.TextSize = 13; lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = card
    
    local sub = Instance.new("TextLabel")
    sub.Size = UDim2.new(0, 200, 0, 16)
    sub.Position = UDim2.new(0, 12, 0, 26)
    sub.BackgroundTransparency = 1; sub.Text = desc; sub.TextColor3 = TEXT_SUB
    sub.Font = Enum.Font.SourceSans; sub.TextSize = 11; sub.TextXAlignment = Enum.TextXAlignment.Left
    sub.Parent = card
    
    local switchBG = Instance.new("TextButton")
    switchBG.Size = UDim2.new(0, 44, 0, 22)
    switchBG.Position = UDim2.new(1, -55, 0.5, -11)
    switchBG.BackgroundColor3 = defaultState and GREY_ACCENT or BUTTON_BG
    switchBG.Text = ""
    switchBG.Parent = card
    Instance.new("UICorner", switchBG).CornerRadius = UDim.new(1, 0)
    
    local switchCircle = Instance.new("Frame")
    switchCircle.Size = UDim2.new(0, 18, 0, 18)
    switchCircle.Position = UDim2.new(0, defaultState and 24 or 2, 0.5, -9)
    switchCircle.BackgroundColor3 = TEXT_MAIN
    switchCircle.Parent = switchBG
    Instance.new("UICorner", switchCircle).CornerRadius = UDim.new(1, 0)
    
    local isOn = defaultState
    switchBG.MouseButton1Click:Connect(function()
        isOn = not isOn
        if isOn then
            TweenService:Create(switchBG, TweenInfo.new(0.2), {BackgroundColor3 = GREY_ACCENT}):Play()
            TweenService:Create(switchCircle, TweenInfo.new(0.2), {Position = UDim2.new(0, 24, 0.5, -9)}):Play()
        else
            TweenService:Create(switchBG, TweenInfo.new(0.2), {BackgroundColor3 = BUTTON_BG}):Play()
            TweenService:Create(switchCircle, TweenInfo.new(0.2), {Position = UDim2.new(0, 2, 0.5, -9)}):Play()
        end
        if callback then callback(isOn) end
    end)
    return card
end

local tabs = {}

local function createTab(name, icon, displayName, order)
    local btn = Instance.new("TextButton")
    btn.LayoutOrder = order
    btn.Size = UDim2.new(1, 0, 0, 36)
    btn.BackgroundColor3 = CARD_BG
    btn.TextColor3 = TEXT_MAIN
    btn.Text = "  " .. icon .. "  " .. displayName
    btn.Font = Enum.Font.GothamSemibold; btn.TextSize = 12; btn.TextXAlignment = Enum.TextXAlignment.Left
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
    
    local stroke = Instance.new("UIStroke"); stroke.Color = GREY_BORDER; stroke.Thickness = 1; stroke.Parent = btn
    
    local indicator = Instance.new("Frame")
    indicator.Size = UDim2.new(0, 3, 0, 20)
    indicator.Position = UDim2.new(0, 4, 0.5, -10)
    indicator.BackgroundColor3 = GREY_ACCENT; indicator.Visible = false; indicator.Parent = btn
    Instance.new("UICorner", indicator).CornerRadius = UDim.new(0, 2)
    
    btn.Parent = sidebarFrame
    
    local page = Instance.new("ScrollingFrame")
    page.Size = UDim2.new(1, 0, 1, 0)
    page.BackgroundTransparency = 1
    page.ScrollBarThickness = 4; page.ScrollBarImageColor3 = GREY_ACCENT
    page.Visible = false
    page.Parent = contentFrame
    
    local pLayout = Instance.new("UIListLayout")
    pLayout.Padding = UDim.new(0, 8); pLayout.Parent = page
    
    tabs[name] = page
    tabButtons[name] = {Button = btn, Stroke = stroke, Indicator = indicator, Layout = pLayout}
    
    btn.MouseButton1Click:Connect(function()
        for tabName, tabData in pairs(tabButtons) do
            local isSelected = (tabName == name)
            tabs[tabName].Visible = isSelected
            tabData.Indicator.Visible = isSelected
            if isSelected then
                tabData.Button.BackgroundColor3 = HOVER_BG; tabData.Stroke.Color = GREY_ACCENT
            else
                tabData.Button.BackgroundColor3 = CARD_BG; tabData.Stroke.Color = GREY_BORDER
            end
        end
    end)
    return page, pLayout
end

local function updateScroll(page, layout)
    page.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 15)
end

local function parseCoordsString(str)
    if not str or str == "" then return nil end
    local x, y, z = string.match(str, "([%d.-]+)%s*,%s*([%d.-]+)%s*,%s*([%d.-]+)")
    if x and y and z then return Vector3.new(tonumber(x), tonumber(y), tonumber(z)) end
    return nil
end

autoFlyToggleBtn.MouseButton1Click:Connect(function()
    isAutoFlyOn = not isAutoFlyOn
    autoFlyToggleBtn.Text = isAutoFlyOn and "Auto Fly: ON" or "Auto Fly: OFF"
    if isAutoFlyOn then
        addLog("Automation: Auto Fly Started")
        task.spawn(function()
            while isAutoFlyOn do
                local didFly = false
                for i = 1, #waypointsData do
                    if not isAutoFlyOn then break end
                    local targetPos = parseCoordsString(waypointsData[i].Input.Text)
                    if targetPos then
                        didFly = true
                        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                            local hrp = player.Character.HumanoidRootPart
                            local spd = tonumber(speedInput.Text) or 500
                            local dist = (hrp.Position - targetPos).Magnitude
                            local tweenInfo = TweenInfo.new(dist / spd, Enum.EasingStyle.Linear)
                            currentTween = TweenService:Create(hrp, tweenInfo, {CFrame = CFrame.new(targetPos)})
                            local bv = Instance.new("BodyVelocity")
                            bv.Velocity = Vector3.new(0,0,0); bv.MaxForce = Vector3.new(9e9,9e9,9e9); bv.Parent = hrp
                            currentTween:Play()
                            currentTween.Completed:Wait()
                            if bv then bv:Destroy() end
                        end
                    end
                end
                if isAutoFlyOn and didFly then task.wait(1.5) elseif not didFly then isAutoFlyOn = false; autoFlyToggleBtn.Text = "Auto Fly: OFF"; addLog("Auto Fly Finished (Queue empty)") break end
            end
        end)
    else
        if currentTween then currentTween:Cancel() end
        addLog("Automation: Auto Fly Stopped by user")
    end
end)
